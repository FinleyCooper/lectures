  \section{Introduction}
  \subsection{Recap of IB Topological Spaces}
  We use algebra to turn topological questions to algebraic questions. This results in a loss of information but lets us prove some results.
  \begin{example}
    \[
	    \R^n \cong \R^m \implies n = m
    \]
  \end{example}
  This is a statement which is very hard to prove otherwise. Likewise we can distinguish topological surfaces with a different number of `holes' in their surface.
  \par
  For this section all functions are continuous until mentioned otherwise.
  \begin{definition}
	  (Map) A \textit{map} is a continuous function.
  \end{definition}
  \begin{lemma}
	  (Gluing lemma) Let $ f:X\to Y $ be a function and $ C, K\subset X$ be closed subsets with $ C\cup K = X $. Then $ f $ is continuous if and only if the restriction of $ f $ on $ C $ and $ K $ is continuous.
  \end{lemma}
  \pf We have that $ f $ is continuous if and only if for all closed sets $ D\subset Y $ the preimage of $ D $ is closed. Since $ \inv f(D) = (\inv f(D)\cap C)\cup(\inv f(D)\cap K) $, this holds if and only if $ \forall D \subset Y $ closed we have that $ \inv f(D) \cap C $ and $ \inv f(D)\cap K $ are closed. Since $ C $ and $ K $ are closed in $ X $, a subset of $ C $ (resp.\ $ K $) is closed in $ C $ (resp.\ $ K $) if and only if it is closed in $ X $. Hence this holds if and only if $ \inv{(f|_C)}(D), \inv{(f|_K)}(D) $ are closed in $ C $ and $ K $ respectively. Hence this holds if and only if the restriction of $ f $ on $ C $ and $ K $ is continuous.\qed
  \begin{lemma}
	  (Lebesgue number lemma) Let $ (X,d) $ be a compact metric space. For any open cover $ \mathcal U = \{U_\alpha\}_{\alpha\in I} $ of $ X $ there exists a $ \delta > 0 $ such that for all $ x\in X $, $ B_\delta(x)\subset U_\alpha $ for some $ \alpha\in I $.
  \end{lemma}
  \pf Suppose no such $ \delta $ existed. Then for all $ n > 0 $ there is some $ x_n\in X $ such that $ B_{1/n}(x_n)\not\subset U_\alpha $ for all $ \alpha\in I $. The sequence $ \{x_n\} $ admits a convergent subsequence due to compactness, so \textit{wlog} replace $ x_n $ with its convergent subsequence, and take $ x_n\to y $. Then take $ \alpha\in I $ such that $ y\in U_\alpha $. Then since $ U_\alpha $ is open there exists a $ r>0 $ such that $ B_r(y)\subset U_\alpha $. Take $ N>0 $ such that
\[
  \begin{cases}
    \frac 1N < \frac r2 \\
    d(x_N,y) < \frac r2
  \end{cases}
\]
which implies that $ B_{1/N}(x_N)\subset B_r(y)\subset U_\alpha $ which is a contradiction.\qed
\subsection{Cell complexes}
\begin{definition}
	(Cell attachment) Given a space $ X $ and a map $ f:S^{n-1} \to X $. The space $ X\cup_f D^n := X \sqcup D^n / \sim $ where $ x\sim f(x), \forall x \in S^{n-1}\subset D^n $ is the result of attaching an $ n $-cell to $ X $ along $ f $.
	\par
	The image of $ f $ is called the \textit{attaching sphere}.
\end{definition}
\begin{definition}
	(Cell complex) A \textit{finite-dimensional cell complex} is obtained by:
	\begin{enumerate}
		\item Start with a set $ X^0 $ with the discrete topology.
		\item Inductively once $ X^{n-1} $ is defined we form $ X^n $ by attaching a collection of $ n $-cells along some collection of attachment maps
			\[
				\{ f_\alpha:S^{n-1} \to X^{n-1}\}_{\alpha\in I}.
			\]
		i.e.\ that $ X^n = \left( X^{n-1} \sqcup \bigsqcup_{\alpha\in I} D_\alpha^n\right) / \sim $ where we define $ x\sim f_\alpha(x) $ for $ x\in S^{n-1}_\alpha = \partial D^n_\alpha $.
	\item $ X=X^k $ for some $ k\ge 0 $. The minimal such $ k $ is called the dimension of $ X $.
	\end{enumerate}
\end{definition}
\begin{remark}
  In general we can make sense of infinite dimensional cell complexes. Suppose that we have
  \[
    X^0 \subset X^1 \subset \cdots
  \]
  such that
  \begin{enumerate}
	  \item $ X^0 $ is discrete,
	  \item $ X^n $ is formed by attaching $ n $-cells to $ X^{n-1} $.
  \end{enumerate}
  Define
  \[
	  X = \bigcup_{n\ge 0} X^n
  \]
  with the topology where $ U \subset X $ is open $ \iff U \cap X^n $ is open in $ X^n $ for all $ n $. We say that $ X $ is a CW-complex and that $ X^n $ is the $ n $-skeleton of $ X $.
 \end{remark}
 \section{Homotopy and the fundamental group}
 \subsection{Homotopy}
 \begin{definition}
	 We denote by $ I $ the closed interval $ [0,1] $.
 \end{definition}
 \begin{definition}
	 (Homotopy) Let $ f,g:X\to Y $. Then we say that a map $ H: X\times I \to Y $ is a \textit{homotopy from} $ f $ \textit{to} $ g $ if $ H(x,0) = f(x) $ and $ H(x,1)=g(x) $. If such an $ H $ exists we say that $ f $ is \textit{homotopic} to $ g $ and $ f\simeq g $.
 \end{definition}
If $ A\subset X $ and $ f|_A = g|_A $ then a homotopy $ H $ from $ f $ to $ g $ is relative to $ A $ (i.e.\ a homotopy $ \rel A $) if $ H(a,t) = f(a) = g(a) $ for all $ a\in A $ and $ t\in I $. If such an $ H $ exists we write $ f\simeq_A g $.
\begin{proposition}
  The relation homotopic $ \rel A $ defines an equivalence relation.
\end{proposition}
\pf
\begin{enumerate}
	\item $ f\simeq_A f $ trivially by $ H(x,t) = f(x) $.
	\item $ f\simeq_A g \implies g\simeq_A f $, by replacing $ t $ with $ 1-t $.
	\item Suppose that $ f\simeq_A g, g\simeq_A h $. Then let $ H_1 $ be the homotopy $ \rel A $ from $ f $ to $ g $ and $ H_2 $ for $ g $ to $ h $. Then define
		\[H(x,t) =
		  \begin{cases}
			  H_1(x,2t) & 0 \le t \le \frac 12 \\
			  H_2(x,2t-1) & \frac 12 \le t \le 1
		  \end{cases}.
		\]
\end{enumerate}
Since $ H_1(x,1) = g(x) = H_2(x,0) $, $ H $ is well-defined, and by the gluing lemma (applied to the closed sets $ X\times[0,\frac12] $ and $ X\times[\frac12,1] $) $ H $ is continuous. Also $ H(a,t) = f(a) $ for all $ a\in A $, so $ H $ is a homotopy $ \rel A $ from $ f $ to $ h $ and therefore $ f\simeq_A h $.\qed
\begin{definition}
	(Homotopy equivalence) Given a map $ f:X\to Y $ we say that $ f $ is a \textit{homotopy equivalence} if there exists a $ g: Y\to X $ such that $ f\circ g \simeq \id_Y $ and $ g \circ f \simeq \id_X $. We call $ g $ a \textit{homotopy inverse} of $ f $. If there exists a homotopy equivalence $ X\to Y $ then $ X $ is \textit{homotopy equivalent} to $ Y $ and we write that $ X\simeq Y $.
\end{definition}
\begin{definition}
	(Contractible) A space $ X $ is \textit{contractible} if $ X $ is homotopy equivalent to the one-point space $ \{*\} $.
\end{definition}
\begin{lemma}
  Let $ f_0,f_1 : X\to Y $ be homotopic maps and $ g_0,g_1 : Y\to Z $ also be homotopic maps. Then
  \[
    g_0 \circ f_0 \simeq g_1 \circ f_1.
  \]
\end{lemma}
\pf We want to show that $ g_0\circ f_0 \simeq g_0\circ f_1 \simeq g_1 \circ f_1 $.\par
Choose homotopies $ H $ from $ f_0 $ to $ f_1 $ and $ H' $ from $ g_0 $ to $ g_1 $. Then $ g_0 \circ H $ is a homotopy from $ g_0f_0 $ to $ g_0f_1 $. We also have that $ H'\circ (f_1 \times \id_I) $ is a homotopy from $ g_0f_1 $ to $ g_1f_1 $ hence we're done by transitivity.\qed
\begin{proposition}
  Homotopy equivalence is an equivalence relation.
\end{proposition}
\pf
\begin{enumerate}
	\item $ X\simeq X $ is obvious from the identity map.
	\item $ X\simeq Y $ then we have $ Y\simeq X $ just by switching the roles of $ f $ and $ g $.
	\item $ X \simeq Y $ and $ Y \simeq Z $. Then suppose $ f,g $ are the homotopy equivalence and its inverse for $ X $ and $ Y $, and $ f',g' $ for $ Y $ and $ Z $. Consider the map $ f'f $ from $ X $ to $ Z $ and its candidate homotopy inverse $ gg' $. By the lemma above,\par
		\begin{align*}
			gg'f'f \simeq g\id_Y f &= gf \simeq \id _X \\
			f'fgg'\simeq f'\id_Y g' &= f'g' \simeq \id_Z
		\end{align*}
		Therefore $ f'f $ is a homotopy equivalence and $ X\simeq Z $.\qed
\end{enumerate}
\begin{example}
  Consider the inclusion
  \[
	  \tau: S^1 \hookrightarrow \R^2\setminus \{0\}
  \]
  with homotopy inverse $ r : \R^2 \setminus \{0\} \to S^1 $ given by projecting each vector onto its unit vector, $ r(x) = x/\|x\| $. Since vector normalisation doesn't do anything to unit vectors one of the directions of composition is obvious: $ r\circ\tau = \id_{S^1} $. For the other direction,
  \[
	  \R^2 \setminus \{0\} \to S^1 \hookrightarrow \R^2 \setminus \{0\}
  \]
  we're given the homotopy
  \begin{align*}
	  H : (\R^2 \setminus \{0\}) \times I &\to \R^2\setminus \{0\}\\
	  (x,t) &\mapsto \frac{x}{t+(1-t)\|x\|}
  \end{align*}
  which is well-defined since the denominator is strictly positive, and is a homotopy from $ \tau \circ r $ to $ \id_{\R^2 \setminus \{0\}} $. Therefore we have that $ S^1 \simeq \R^2 \setminus \{0\} $. Similarly we have that $ S^{n-1} \simeq \R^n \setminus \{0\}$.
\end{example}
\begin{example}
  We have that $ \R^n $ is contractible. Let $ f:\R^n\to\{0\} $ be the constant map and $ g:\{0\}\hookrightarrow\R^n $ be the inclusion. Then
  \[
	  f\circ g = \id_{\{0\}}
  \]
  so one direction is trivial. For the other direction, $ g\circ f:\R^n\to\R^n $ is the constant map $ x\mapsto 0 $. Define the straight-line homotopy
  \begin{align*}
	  H : \R^n \times I &\to \R^n\\
	  (x,t) &\mapsto tx.
  \end{align*}
  This is continuous since scalar multiplication $ \R\times\R^n\to\R^n $ is continuous. We have $ H(x,0) = 0 = (g\circ f)(x) $ and $ H(x,1) = x = \id_{\R^n}(x) $, so $ H $ is a homotopy from $ g\circ f $ to $ \id_{\R^n} $. Therefore $ g\circ f\simeq\id_{\R^n} $, so $ f $ is a homotopy equivalence with homotopy inverse $ g $. Hence $ \R^n\simeq\{0\} $, i.e.\ $ \R^n $ is contractible.
  \par
  The same argument works for any $ X\subset\R^n $ which is star-shaped about some point $ x_0 $ (e.g.\ any convex set), using $ H(x,t) = (1-t)x_0 + tx $.
\end{example}
\begin{definition}
(Deformation Retraction) Let $ i:A\hookrightarrow X $ be the inclusion of a subspace. We say that $ A $ is a \textit{deformation retract} of $ X $ if there is a map $ r:X\to A $ such that $ r\mid_A=\id_A $ and $ i\circ r\simeq\id_X $. If the homotopy is relative to $ A $, then $ A $ is a strong deformation retract of $ X $.
\end{definition}
\begin{remark}
  This implies that $ X\simeq A $ by the inclusion map from $ A $ to $ X $ and $ r $ from $ X $ to $ A $. This is a strictly stronger condition.
\end{remark}
\begin{definition}
	(Retraction) A \textit{retraction} of $ X $ to $ A $ is a map $ r:X\to A $ such that $ r\mid_A = \id $.
\end{definition}
\begin{remark}
  A retraction need not be a homotopy equivalence.
\end{remark}
\begin{example}
	$ X $ is a space and $ x\in X $. Then $ X $ retracts to $ \{x\} $. But $ X\simeq \{x\} \iff X $ is contractible
\end{example} 
\subsection{Path}
\begin{definition}
	(Path) For a space $ X $ and points $ x_0,x_1\in X $, a \textit{path} from $ x_0 $ to $ x_1 $ is a map $ \gamma:I \to X $ such that $ \gamma(0) = x_0 $ and $ \gamma(1) = x_1 $. 
\end{definition}
We write that $ \gamma: x_0 \rightsquigarrow x_1 $, and if $ x_0 =x_1 $ we say that $ \gamma $ is a loop based at $ x_0 $.\par
If we have $ \gamma:x_0 \rightsquigarrow x_1 $ and $ \gamma':x_1\rightsquigarrow x_2 $ then the concatenation of $ \gamma\circ \gamma' $ is defined by using the gluing lemma and going in double time which goes from $ x_0 $ to $ x_2 $.\par
We also define the inverse path of $ \gamma: x_0\rightsquigarrow x_1 $ as $ \inv \gamma(t) = \gamma(1-t): x_1\rightsquigarrow x_0 $.\par
Finally we define the constant path $ c_{x_0}(t) = x_0: x_0\rightsquigarrow x_0 $.\par
If we define a relation $ \sim $ on $ X $ by $ x\sim y \iff $ there is a path from $ x $ to $ y $, these definitions show that $ \sim $ is an equivalence relation.
\begin{definition}
	(Path components) The equivalence classes of $ \sim $ are called \textit{path components} of $ X $. The set of all path components of $ X $ is denoted by $ \pi_0(X) $. If $ |\pi_0(X)| = 1 $ then we say that $ X $ is \textit{path-connected}.
\end{definition}
\begin{remark}
  $ X $ is path-connected $ \iff\; \forall x,y\in X $ there exists a path $ x\rightsquigarrow y $.
\end{remark}
\begin{proposition}
  Given a map $ f:X\to Y $ there is a well-defined associated function 
  \[
    \pi_0(f): \pi_0(X) \to \pi_0(Y)
  \]
  given by $ \pi_0(f)[x] = [f(x)] $. Futhermore,
  \begin{enumerate}
	  \item If $ f\simeq g $, then $ \pi_0(f) = \pi_0(g) $;
	  \item For maps $ A \overset h\rightarrow B \overset k\rightarrow C $ we have that 
		  \[
		    \pi_0(k\circ h) = \pi_0(k) \circ \pi_0(h).
		  \]
	  \item $ \pi_0(\id_X) = \id_{\pi_0(X)} .$
  \end{enumerate}
\end{proposition}
\pf Choose $ x,x'\in X $ such that $ [x]=[x'] $. Then there exists a path $ \gamma $ from $ x $ to $ x' $ in $ X $. Now $ f\circ \gamma $ is a path from $ f(x) $ to $ f(x') $ hence they are in the same path-component so $ \pi_0(f) $ is well-defined.\par
For (i), let $ H:X\times I \to Y $ be a homotopy from $ f $ to $ g $. Choose any $ x\in X $. The map $ H(x,\cdot) $ is a path from $ f(x) $ to $ g(x) $ so $ [f(x)] = [g(x)] $ and hence $ \pi_0(f) = \pi_0(g) $. (ii) and (iii) are obvious.\qed
\begin{corollary}
  If $ f:X\to Y $ is a homotopy equivalence then $ \pi_0(f) $ is bijective.
\end{corollary}
\pf Let $ g: Y \to X $ be the homotopy inverse of $ f $ so $ fg\simeq \id_Y $ hence $ \pi_0(f) \pi_0(g) = \pi_0(\id_Y) = \id_{\pi_0(Y)} $. Similary $ gf\simeq \id_X $ gives the other direction. Hence $ \pi_0(f) $ is a bijection.\qed
\begin{example}
	Consider the space $ \{0,1\} $ equipped with the discrete topology. Let's prove it is not contractible. If it were, then we'd have $ \pi_0(\{0,1\}) \simeq \pi_0(*) $ (the one point space). This is false because $ \{0,1\} $ has two path-components.
\end{example}
\begin{example}
	$ \R^2 $ does not retract to $ A = \{(-1,0), (1,0)\} $. Suppose that $ r: \R^2 \to A $ is a retraction. Then the composition
	\[
		\pi_0(A) \overset{\pi_0(\tau)}\rightarrow \pi_0(\R^2) \overset{\pi_0(r)}\rightarrow \pi_0(A)
	\]
	has to be the identity. This is impossible.
\end{example}
\begin{definition}
	(Homotopic paths) The paths $ \gamma,\gamma':I \to X $ are \textit{homotopic as paths} if they are homotopic $ \rel \{0,1\} \subseteq I$.
\end{definition}
\begin{lemma}
  If $ \gamma_0,\gamma_1:  x_0\rightsquigarrow x_1 $ and $ \gamma_0', \gamma_1':x_1 \rightsquigarrow x_2 $ are homotopic then $ \gamma_0\circ\gamma_0' \simeq \gamma_1 \circ \gamma_1':x_0\rightsquigarrow x_2 $.
\end{lemma}
\pf Let $ H $ be the homotopy of paths from $ \gamma_0 $ to $ \gamma_1 $ and let $ H' $  be the homotopy of paths from $ \gamma_0' $ to $ \gamma_1' $. Define 
\[
  H''(s,t) = \begin{cases}
	  H(2s,t) & 0\le s \le \frac 12 \\
	  H'(2s-1,t) & \frac 12 \le s \le 1
  \end{cases}
\]
which is continuous by the gluing lemma. At $ H''(s,0) = (\gamma_0\circ \gamma_0')(s) $ and $ H''(s,1) = \gamma_1\circ\gamma_1' $. So $ H'' $ is a homotopy from $ \gamma_0\circ \gamma_0' $ to $ \gamma_1 \circ\gamma_1' $.
\begin{proposition}
  Let $ \gamma_0: x_0\rightsquigarrow x_1, \gamma_1:x_1\rightsquigarrow x_2, \gamma_2:x_2\rightsquigarrow x_3 $ be paths in $ X $. Then
  \begin{enumerate}
	  \item $ (\gamma_0 \circ \gamma_1) \circ \gamma_2 \simeq \gamma_0 \circ (\gamma_1 \circ \gamma_2) $;
	  \item $ \gamma_0 \circ c_{x_1} \simeq \gamma_0 \simeq c_{x_0} \circ \gamma_0 $;
	  \item $ \gamma_0 \inv\gamma_0 \simeq c_{x_0} $, $ \inv \gamma_0\circ \gamma_0\simeq c_{x_1} $.
  \end{enumerate}
\end{proposition}
\pf
From these, we arrive at the following formulae.

\begin{enumerate}
    \item[(i)] The homotopy
    \[
    H(s,t)=
    \begin{cases}

        \gamma_1(4s-1-t)
        & \dfrac{t+1}{4}\leq s\leq \dfrac{t+2}{4},\\[8pt]

        & \dfrac{t+2}{4}\leq s\leq 1.
    \end{cases}
    \]
    goes from $(\gamma_0\cdot\gamma_1)\cdot\gamma_2$
    to $\gamma_0\cdot(\gamma_1\cdot\gamma_2)$ as paths
    from $x_0$ to $x_3$.

    \item[(ii)] The homotopy
    \[
    H(s,t)=
    \begin{cases}
        \gamma_0\left(\dfrac{2s}{t+1}\right)
        & 0\leq s\leq \dfrac{t+1}{2},\\[8pt]
        x_1
        & \dfrac{t+1}{2}\leq s\leq 1,
    \end{cases}
    \]
    goes from $\gamma_0\cdot c_{x_1}$ to $\gamma_0$ as paths
    from $x_0$ to $x_1$.

    \item[(iii)] The homotopy
    \[
    H(s,t)=
    \begin{cases}
        \gamma_0(2s)

        & \dfrac{1-t}{2}\leq s\leq \dfrac{1+t}{2},\\[8pt]
        \gamma_0(2-2s)
        & \dfrac{1+t}{2}\leq s\leq 1.
    \end{cases}
    \]
    goes from $\gamma_0\cdot\overline{\gamma_0}$
    to $c_{x_0}$ as paths from $x_0$ to $x_0$.
\end{enumerate}
\begin{proposition}
  Let $ X $ be a space, $ x_0 \in X $. Define
  \[
	  \pi_1(X,x_0) = \{ \text{loops in } X \text{ based at } x_0\}/ \text{homotopy}.
  \]
  Then the rule $ [\gamma]\circ [\gamma'] = [ \gamma \circ \gamma'] $ defines a group structure on $ \pi_1(X,x_0) $. The identity is paths homotopic to the constant path.
\end{proposition}
\pf The previous proposition.
\begin{definition}
	(Fundamental Group) We call $ \pi_1(X,x_0) $ the \textit{fundamental group} of $ X $ at $ x_0 $.
\end{definition}
\begin{definition}
	(Based space) A \textit{based space} is a pair $ (X,x_0) $ where $ x_0 $ is called the \textit{basepoint}. Given based spaces $ (X,x_0), (Y,y_0) $ a \textit{based map} between them is a map $ f:X\to Y $ such that $ f(x_0) = y_0 $. A \textit{based homotopy} is a homotopy relative to the basepoint.
\end{definition}
\begin{proposition}
  For a based map $ f: (X,x_0) \to (Y,y_0) $ there is a well-defined associated group homomorphism
  \[
    \pi_1(f): \pi_1(X,x_0) \to \pi_1(Y,y_0)
  \]
  given by $ \pi_1(f)[\gamma] \to [f \circ \gamma] $, with the following
  \begin{enumerate}
	  \item If $ f $ is based homotopic to $ g $ then $ \pi_1(f) = \pi_1(g) $.
	  \item $ \pi_1(f\circ g) = \pi_1(f) \circ \pi_1(g) $.
	  \item $ \pi_1(\id_X) = \id_{\pi_1(X,x_0)} $.
  \end{enumerate}
\end{proposition}
\pf $ \pi_1(f) $ is well-defined because if $ \gamma \simeq \gamma' $ then $ f\circ \gamma \simeq f \circ\gamma' $.\par
We know that $ f \circ c_{x_0} = c_{f(x_0)} = c_{y_0} $ so $ \pi_1(f) $ preserves the identity element of the group. We also have that 
\[
  f\circ (\gamma \cdot \gamma') = (f \circ \gamma) \cdot (f \circ \gamma')
\]
so $ \pi_1(f) $ is multiplicative, hence $ \pi_1(f) $ is a group homomorphism.
\par
For (i) if $ f,g $ are homotopic $ \rel x_0 $ then $ f\circ \gamma \simeq g \circ \gamma $ so $ \pi_1(f) = \pi_1(g) $. For (ii) since $ \pi_(f\circ g)[\gamma] = [f \circ g\circ \gamma] = \pi_(f)[g\circ \gamma] = \pi_1(f) \circ \pi_1(g) $. Part (iii) is obvious. \qed
\par
For ease of notation, we'll just write $ f_* $ to mean $ \pi_1(f) $.
\begin{proposition}
  Let $ u: x_0\rightsquigarrow x_1 $ be a path in $ X $. Then it induces a group homomorphism 
\begin{align*}
	u_\#: \pi_1(X,x_0) &\to \pi_1(X,x_1) \\
	[\gamma] &\to [\inv u \cdot \gamma \cdot u]
\end{align*}
satisfying
\begin{enumerate}
	\item $ u \simeq u' \implies u_\# = u_\#' $;
	\item $ (c_{x_0})_\# = \id_{\pi_1(X,x_0)} $;
	\item $ v: x_1\rightsquigarrow x_2\implies (u\cdot v)_\# = v_\# u_\# $;
	\item If $ x_0 = x_1 $, then $ u_\# $ is conjugation by $ [u] $;
	\item $ f:X\to Y $, then we have
		\[
			(f\circ u)_\# \circ f_* = f_* \circ u_\#.
		\]
\end{enumerate}
\end{proposition}
\pf Parts (i)-(iv) are obvious. We'll just prove (v).\par
We have that 
\begin{align*}
	((f\circ u)_\# \circ f_*)[\gamma] &= (f\circ u)_\#[f\circ \gamma] \\
					  &= [(f\circ u)^{-1} \cdot (f\circ \gamma) \cdot (f\circ u)]\\
					  &= [f\circ(\inv u \cdot \gamma \cdot u)] \\
					  &= f_*[\inv u \cdot \gamma \cdot u] \\
					  &= f_* u_\#[\gamma].\qed
\end{align*}
Suppose $ f,g:X\to Y $ with $ x_0\in X $ and $ H:X\times I \to Y $ is a homotopy from $ f $ to $ g $. Then $ H(x_0, \cdot): I \to Y $ is a path from $ f(x_0) $ to $ g(x_0) $. Write this path as $ u $. This sets us up to prove the following lemma
\begin{lemma}
  For $ u,f,g $ given as above,
  \[
    u_\# \circ f_*  = g_*.
  \]
\end{lemma}

\begin{align*}

\begin{theorem}
	Let $ f: X\to Y $ and choose $ x_0\in X $. Then $ f_*: \pi_1(X,x_0) \to \pi_1(Y,f(x_0)) $. If $ f $ is a homotopy equivalence, then $ f_* $ is a group isomorphism. 
\end{theorem}

\[
  u'(t) = H'(x_0,1-t): I \to X.
\]

We know that $ u'_\# $ is an isomorphism hence $ g_* \circ f_* $ and $ f_*\circ g_* $ are both isomorphisms (by doing the same thing the otherway around). \begin{definition}
	(Simply connected) A space $ X $ is \textit{simply connected} if it is path-connected and satifies $ \pi_1(X,x_0) = 1 $ for all $ x_0\in X $.
\end{definition}
\begin{remark}
  From the previous proposition, since $ X $ is path-connected, we can connected every two points with a path, so $ \pi_1(X,x_0) \cong \pi_1(X,x_1) $, hence it's equivalent that $ \pi_1(X,x_0) =1 $ for a $ x_0\in X $ is enough for equivalence of simply connectedness.
\end{remark}
\begin{example}
	Any contractible space is simply connected. This is clear since if $ X $ is contractible then $ X\simeq \{*\} $ so $ \pi_1(X,x_0)\cong \pi_1(*,*) = 1 $.
\end{example}
\begin{lemma}
  A space $ X $ is simply connected if and only if for all $ x_0,x_1 $ there is a unique homotopy class of paths from $ x_0 $ to $ x_1 $.
\end{lemma}
\pf The forward direction is obvious just from taking $ x_0 = x_1 $ where the result follows directly.\par
Instead if we suppose that $ X $ is simply connected and $ \pi_1(X,x_1) = 1 $. Since $ X $ is path-connected, there is a path from $ x_0 $ to $ x_1 $. Choose two paths $ \gamma $ and $ \gamma' $ from $ x_1 $ to $ x_1 $. Then $ [\inv \gamma \cdot \gamma']\in \pi_1(X,x_1) $. So $ \inv\gamma\cdot \gamma'\simeq c_{x_1} $. Hence there is a homotopy from $ \gamma \cdot \inv \gamma \cdot \gamma' $ to $ \gamma \cdot c_{x_1}	$. Since we can reduce the path $ \gamma \cdot \inv \gamma $ to a point, we have that $ \gamma' \simeq \gamma $.\qed
\section{Covering Space}
\begin{definition}
	(Covering map) A map $ f:\tilde X\to X $ is a \textit{covering map} if it is surjective and for all $ p\in X $ there is an open neighbourhood $ U_p\subset X $ such that $ \inv f (U_p) \simeq \bigsqcup_{\alpha\in I} V_\alpha $ where each 
	\[
		f |_{V_\alpha}:V_\alpha \to U_p
	\]
	is a homeomorphism. We say that $ \tilde X $ is the covering space of $ X $.
\end{definition}
\begin{example}
  Clearly homeomorphism are covering maps
\end{example}
\begin{example}
  If $ p:\tilde X \to X $ and $ q:\tilde Y \to Y $ are covering maps then $ p\times q: \tilde X \times \tilde Y \to X\times Y $ is a covering map.
\end{example}
\begin{example}
	Consider $ S^1 = \{z \in \C : |z| = 1 \}\subseteq \C $. Take $ p(t) = e^{2\pi i t}:\R\to S^1 $.\par
	To prove that $ p $ is a covering map, consider the set $ U_{y>} = \{ x+iy \in S^1 : y> 0\} $, the open upper semicircle. Then
	\begin{align*}
		\inv p(U_{y>0}) = \bigsqcup_{j\in \Z} \left(j,j+\frac 12\right) \subset \R
	\end{align*}
	which is the disjoint union of $ \Z $ many open intervals. Furthermore 
	\[
		p|_{(j,j+\frac 12)}: \left(j,j+\frac 12\right) \to U_{y>0}
	\]
	is a continuous bijection with inverse
	\begin{align*}
		U_{y>0} &\longrightarrow \left(j,j+\frac 12\right) \\
		x+ iy &\longrightarrow j+\inv\cos(x)/2\pi.
	\end{align*}
	Similarly with $ U_{y<0}, U_{x>0}, U_{x<0} $ these open sets cover $ S^1 $. Thus $ p $ is a covering map.
\end{example}
\begin{example}
	Consider $ p(z) = z^n: S^1 \to S^1 $ and denote $ \eta = e^{\frac{2\pi i}n} $ and let $ \zeta $ be some $ n $th root of $ y $. Then we have that
	\[
		\inv p(y) = \{\zeta, \eta \zeta, \eta^2\zeta,\dots, \eta^{n-1}\zeta\}
	\]
	and $ \inv p(S^1 \setminus \{y\}) $ is the complement of this set. Now we can define
	\[
		V_0 = \{z\in S^1 \mid 0 < \arg(z/\eta) < 2\pi /n\}
	\]
	and $ V_i = V_0 \cdot \eta^i $ for $ 0\le i < n $, so we get that
	\[
		\inv p(S^1 \setminus \{y\}) = \bigsqcup_{i=0}^{n-1} V_i.
	\]
	Then $ p\mid_{V_i} : V_i\to S^1 \setminus \{y\} $ is a homeomorphism and
	\[
		S^1 =\bigcup_{y\in S^1} S^1 \setminus \{y\}
	\]
	therefore $ p $ is a covering map.
\end{example}
\begin{example}
	Consider $ S^2 = \{ v\in \R^3 \mid ||v|| = 1 \} \subseteq \R^3 $. We can define an equivalence relation $ x\sim -x $ on $ S^2 $ which defines a quotient space $ \RP^2 = S^2 / \sim $. Consider the quotient map
	\begin{align*}
		p: S^2 &\longrightarrow \RP^2 \\
	  x & \longrightarrow [x]
	\end{align*}
	which we claim is a covering map. Let $ V = \{(x,y,z|) \in S^2 \mid z \ne 0\} $ then $ V\simeq V_{z> 0} \sqcup V_{z<0} $ so we can define $ p $ on the restriction to each of these subspaces. We claim that
	\begin{align*}
		p\mid_{V_{z>0}}: V_{z>0} &\to p(V) \\
		p\mid_{V_{z<0}}: V_{z<0} & \to p(V)
	\end{align*}
	are homeomorphisms. Similarly define $ V_x,V_y\subset S^2 $ then $ p(V_x), p(V_y), p(V_z) $ are evenly covered and forms san open cover of $ \RP^2 $. Hence $ p $ is a covering map.
\end{example}
\begin{definition}
	(Lifting) Let $ :p: \tilde X \to X $ be a covering map and $ f: Y \to X $ be any map. Then a map $ \tilde f:Y \to \tilde X $ is a \textit{lift} of $ f $ along $ p $ if $ f = p \circ \tilde f $.
\end{definition}
\begin{lemma}
	(Uniqueness of lifts lemma) Let $ p: \tilde X \to X $ be a covering map and $ f: Y \to X $ be any map. Let $ \tilde f_0, \tilde f_1 $ be two lifts of $ f $ along $ p $. Then the set $ \{y\in Y : \tilde f_0(y) = \tilde f_1(y)\} $ is clopen in $ Y $. Consequently if $ Y $ is connected then $ S = Y $ or $ S = \emptyset $.
\end{lemma}
\pf First we'll show that $ S $ is open. Choose $ y\in S $ and let $ U $ be an open neighbourhood of $ f(s) \in X $ which is evenly covered, so $ \inv p(U) = \bigsqcup_{\alpha \in I} V_\alpha $. Then $ \tilde f_0(y) $ and $ \tilde f_1(y) $ are equal so lie in the same $ V_\beta $. Thus we have that on $ N = \tilde f_0^{-1}(V_\beta) \cap \tilde f_1^{-1}(V_\beta) $ we have that 
\[
	p|_{V_\beta} \circ \tilde f_0 \mid_N = f\mid_N = p\mid_{V_\beta} \circ \tilde f_1 \mid_N
\]
but since $ p\mid V_\beta $ is a homeomorphism, it has an inverse so $ \tilde f_0\mid_N = \tilde f_1\mid_N $ and $ y\in N \subset S $, so $ S $ is open.\par
Now we have to show that $ S $ is closed. We'll show that $ Y\setminus S $ is open. Suppose that $ y\notin S $, so $ \tilde f_0(y) \ne \tilde f_1(y) $. Choose an evenly covered open neighbourhood $ U $ at $ f(y) $ and identify $ \inv p(U) = \bigsqcup_{\alpha \in I} V_\alpha  $ so $ \tilde f_0(y) \in V_\alpha $ and $ \tilde f_1(y) \in V_\beta $ for some $ \alpha,\beta\in I $ since $ V_\alpha $ is mapped homeomorphically to $ U $ and the covering maps differ, we must have that $ \alpha\ne \beta $. So $ N = \tilde f_1^{-1}(V_\beta) \cap \tilde f_0^{-1}(V_\alpha) $ is an open neighbourhood of $ y $ so must intersect $ S $ by the definition of closure. But then $ V_\alpha $ and $ V_\beta $ must intersect which is a contradiction.

\begin{lemma}
	(Homotopy lifting lemma) Let $ p:\tilde X \to X $ be a covering map, let $ Y $ be an arbitrary topological space, and let $ H: Y\times I \to X $ be a homotopy from $ f_0 $ to $ f_1 $, i.e.\ $ H(-,0)=f_0 $ and $ H(-,1)=f_1 $. Let $ \tilde f_0 : Y \to \tilde X $ be a lift of $ f_0 $. Then there exists a unique homotopy $ \tilde H : Y \times I \to \tilde X $ such that
	\begin{enumerate}
		\item $ \tilde H(-,0) = \tilde f_0 $;
		\item $ \tilde H $ is a lift of $ H $ along $ p $, i.e.\ $ p\circ\tilde H = H $.
	\end{enumerate}
	In particular, $ \tilde H $ is a homotopy from $ \tilde f_0 $ to the lift $ \tilde f_1 := \tilde H(-,1) $ of $ f_1 $.
\end{lemma}
\pf Throughout, $ I=[0,1] $. We use the \emph{uniqueness of lifts} lemma in the following form: if $ Z $ is a connected space and $ g_1,g_2:Z\to\tilde X $ are two lifts along $ p $ of the same map $ Z\to X $ which agree at one point of $ Z $, then $ g_1=g_2 $. We only ever apply it with $ Z=I $; in particular no connectedness (or local path connectedness) of $ Y $ is needed anywhere below.
\par
\textbf{Step 1: uniqueness.} Let $ \tilde H_1 $ and $ \tilde H_2 $ both satisfy (1) and (2). Fix $ y\in Y $. The paths $ t\mapsto \tilde H_1(y,t) $ and $ t\mapsto \tilde H_2(y,t) $ in $ \tilde X $ are both lifts of the path $ t\mapsto H(y,t) $ in $ X $, and by (1) both start at $ \tilde f_0(y) $. Since $ I $ is connected, uniqueness of lifts gives $ \tilde H_1(y,t)=\tilde H_2(y,t) $ for all $ t\in I $. As $ y\in Y $ was arbitrary, $ \tilde H_1=\tilde H_2 $.
\par
This argument works for every space $ Y $, so it applies in particular to any open subset $ Y'\subseteq Y $ with the restricted data $ H|_{Y'\times I} $ and $ \tilde f_0|_{Y'} $.
\par
\textbf{Step 2: local existence.} We claim that for every $ y_0\in Y $ there exist an open neighbourhood $ U_{y_0} $ of $ y_0 $ in $ Y $ and a continuous map $ \tilde H_{y_0}:U_{y_0}\times I\to\tilde X $ such that $ p\circ\tilde H_{y_0}=H|_{U_{y_0}\times I} $ and $ \tilde H_{y_0}(-,0)=\tilde f_0|_{U_{y_0}} $.
\par
\emph{(2a) Choosing a partition.} Fix $ y_0\in Y $. For each $ t\in I $ the point $ H(y_0,t) $ has an evenly covered open neighbourhood in $ X $. By continuity of $ H $ there are an open neighbourhood $ W_t $ of $ y_0 $ in $ Y $ and an open neighbourhood $ N_t $ of $ t $ in $ I $ such that $ H(W_t\times N_t) $ is contained in an evenly covered open subset of $ X $. The sets $ \{N_t\}_{t\in I} $ form an open cover of the compact space $ I $, so there is a finite subcover $ N_{s_1},\dots,N_{s_k} $. Put
\[
	W = W_{s_1}\cap\cdots\cap W_{s_k},
\]
an open neighbourhood of $ y_0 $. By the Lebesgue number lemma there is an $ N\ge 1 $ such that each interval $ [\frac iN,\frac{i+1}N] $ lies in some $ N_{s_j} $. Write $ t_i=\frac iN $. Then for each $ i=0,\dots,N-1 $ there is an evenly covered open subset $ U_i\subseteq X $ with
\[
	H\left(W\times[t_i,t_{i+1}]\right)\subseteq H(W_{s_j}\times N_{s_j})\subseteq U_i .
\]
Write
\[
	\inv p(U_i) = \bigsqcup_{\alpha\in A_i} V_{i,\alpha}, \qquad p\mid_{V_{i,\alpha}}:V_{i,\alpha} \overset{\sim}{\longrightarrow} U_i
\]
where the $ V_{i,\alpha} $ are pairwise disjoint open subsets of $ \tilde X $ and $ \overset\sim\longrightarrow $ means a homeomorphism.
\par
\emph{(2b) Inductive construction with shrinking.} We show by induction on $ i=0,\dots,N $ that there exist an open neighbourhood $ W_i' $ of $ y_0 $ with $ W_i'\subseteq W $ and a continuous map $ G_i:W_i'\times[0,t_i]\to\tilde X $ such that $ p\circ G_i=H|_{W_i'\times[0,t_i]} $ and $ G_i(y,0)=\tilde f_0(y) $ for all $ y\in W_i' $.
\par
For $ i=0 $ take $ W_0'=W $ and $ G_0(y,0)=\tilde f_0(y) $; then $ p\circ G_0=f_0|_W=H(-,0)|_W $.
\par
Suppose $ W_i' $ and $ G_i $ are constructed for some $ i<N $. The point $ G_i(y_0,t_i) $ satisfies $ p(G_i(y_0,t_i))=H(y_0,t_i)\in U_i $, so it lies in $ \inv p(U_i) $, hence in exactly one sheet $ V=V_{i,\alpha} $. Since $ y\mapsto G_i(y,t_i) $ is continuous and $ V $ is open,
\[
	W_{i+1}' = \{y\in W_i' \mid G_i(y,t_i)\in V\}
\]
is an open neighbourhood of $ y_0 $. Define $ G_{i+1}:W_{i+1}'\times[0,t_{i+1}]\to\tilde X $ by
\[
	G_{i+1}(y,t)=
	\begin{cases}
		G_i(y,t) & t\in[0,t_i],\\
		\left(p\mid_{V}\right)^{-1}\bigl(H(y,t)\bigr) & t\in[t_i,t_{i+1}].
	\end{cases}
\]
The second formula makes sense because $ H(y,t)\in U_i $ for $ y\in W\supseteq W_{i+1}' $ and $ t\in[t_i,t_{i+1}] $. The two formulas agree at $ t=t_i $: indeed $ G_i(y,t_i)\in V $ and $ p(G_i(y,t_i))=H(y,t_i) $, so $ \left(p\mid_V\right)^{-1}(H(y,t_i))=G_i(y,t_i)$. Both pieces are continuous on closed subsets $ W_{i+1}'\times[0,t_i] $ and $ W_{i+1}'\times[t_i,t_{i+1}] $ whose union is the whole domain, so $ G_{i+1} $ is continuous by the pasting lemma. Clearly $ p\circ G_{i+1}=H $ and $ G_{i+1}(-,0)=\tilde f_0 $ on $ W_{i+1}' $, which completes the induction.
\par
Finally set $ U_{y_0}=W_N' $ and $ \tilde H_{y_0}=G_N $. Note that we shrank the neighbourhood only finitely many times, so $ U_{y_0} $ is still an open neighbourhood of $ y_0 $; this is exactly where one would otherwise want $ Y $ to be (locally) connected, and the shrinking trick avoids it.
\par
\textbf{Step 3: the local lifts are compatible.} Let $ y_0,y_1\in Y $ and $ Y'=U_{y_0}\cap U_{y_1} $, which is open. If $ Y'=\emptyset $ there is nothing to show. Otherwise $ \tilde H_{y_0}|_{Y'\times I} $ and $ \tilde H_{y_1}|_{Y'\times I} $ are both lifts of $ H|_{Y'\times I} $ whose restriction to $ Y'\times\{0\} $ is $ \tilde f_0|_{Y'} $. By the uniqueness in Step 1 (applied to $ Y' $), $ \tilde H_{y_0}=\tilde H_{y_1} $ on $ (U_{y_0}\cap U_{y_1})\times I $.
\par
\textbf{Step 4: gluing.} Define $ \tilde H:Y\times I\to\tilde X $ by
\[
	\tilde H(y,t)=\tilde H_{y_0}(y,t)\quad\text{for any } y_0\in Y \text{ with } y\in U_{y_0}.
\]
(Such a $ y_0 $ exists, e.g.\ $ y_0=y $.) This is well defined by Step 3. The sets $ U_{y_0}\times I $ form an open cover of $ Y\times I $ and $ \tilde H $ restricts to the continuous map $ \tilde H_{y_0} $ on each of them, so $ \tilde H $ is continuous, since continuity is a local property. Moreover $ p\circ\tilde H=H $ and $ \tilde H(-,0)=\tilde f_0 $ hold on each $ U_{y_0}\times I $, hence on $ Y\times I $. This proves existence, and uniqueness was shown in Step 1.\qed

\end{corollary}

Suppose we have $ p:\tilde X \to X $ being a covering map and a point $ x_0\in X $ with $ \tilde x_0\in \inv p(x_0) $. For a path $ u: x_0\rightsquigarrow x_1 $ in $ X $, the first corollary tells us that there exists a lift $ \tilde u $ of $ u $ with domain $ \tilde x_0 $. \[
  u_*: \inv p(x_0) \to \inv p(x_1).

\begin{corollary}
	We have the following:
	\begin{enumerate}
		\item If $ u:x_0\rightsquigarrow x_1 $ and $ u':x_1\rightsquigarrow x_2 $ then $ (u\cdot u')_* = u_*' \circ u_* $;
		\item $ u_* $ is a bijection;
		\item If $ X $ is path connected then $ |\inv p(x_0)| = |\inv p(x_1)| $.
	\end{enumerate}
\end{corollary}
\pf We can see that (ii) and (iii) following from (i), so we'll just prove (i). Choose a lift $ \tilde u:\tilde x_0\to \tilde x_1 $ of $ u $ and $ \tilde u': \tilde x_1 \to \tilde x_2 $ of $ u' $. Then $ \tilde u \cdot \tilde u' : \tilde x_0\to\tilde x_2 $ is a lift of $ u\cdot u' $. Thus $ (u\cdot u')_*(\tilde x_0) = \tilde x_2 = u_*'(\tilde x_1) = u_*'(u_*(\tilde x_0)) = (u_*'\circ u_*)(\tilde x_0) $.\qed
\begin{lemma}
  Let $ p:\tilde X\to X $ be a covering map. Choose $ x_0\in X $, $ \tilde x_0\in \inv p(x_0) $. Then
  \[
    p_*: \pi_1(\tilde X,\tilde x_0) \to \pi_1(X,x_0)
  \]
  is injective.
\end{lemma}
\pf Choose a loop $ \gamma $ in $ \tilde X $, based at $ \tilde x_0 $ such that $ [\gamma]\in\ker p_* $. Hence $ [p\circ \gamma] = 0 $. So there is a homotopy $ H: I\times I\to X $ from $ p\circ\gamma $ to $ c_{x_0} $. By the homotopy lifting lemma, $ H $ admits a lift $ \tilde H $ which is a homotopy $ \gamma $ to some lift of $ c_{x_0} $. By uniqueness of lifts, the lift of $ c_{x_0} $ is $ c_{\tilde x_0} $ so $ [\gamma] = [c_{\tilde x_0}] = 0\in \pi_1(\tilde X,\tilde x_0) $ so $ p_* $ is injective.

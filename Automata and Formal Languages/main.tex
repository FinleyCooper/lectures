  \section{Introduction}
  In this course we will have $ 0\in \N $.
\begin{definition}
	(Alphabet) An \textit{alphabet} $ \Sigma $ is a finite non-empty set with elements which are \textit{symbols} or \textit{letters}.
\end{definition}
\begin{definition}
	(String) A \textit{string} or \textit{word} of length $ m $ over $ \Sigma $ is $ \alpha = x_1x_2\dots x_m $ with $ x_i\in \Sigma $ or more formally $ \alpha \in\Sigma^m $. We write $ |\alpha| = m $ for the length of $ \alpha $.
\end{definition}
\begin{remark}
  We will denote the empty word as $ \varepsilon $ and denote
  \[
	  \Sigma^* = \bigcup_{n\in \N}\Sigma^n
  \]
  and $ \Sigma^\dagger = \Sigma^* \setminus \{\varepsilon\} $.
\end{remark}
\begin{definition}
	(Concatenation) If $ \alpha = x_1\cdots x_m\in \Sigma^m $ and $ \beta= y_1\cdots y_n\in \Sigma^n $ then $ \alpha\beta $ is $ x_1\cdots x_my_1\cdots y_n \in \Sigma^{m+n} $.
\end{definition}
For $ 0\le i \le j \le m $, $ x_{i+1}\cdots x_j $ is a substring of $ \alpha $ of length $ j-i $. So for $ \alpha,\beta\in \Sigma^* $, $\beta $ is a substring of $ \alpha $ if and only if there exists $ \gamma,\delta\in\Sigma^* $ with $ \alpha = \gamma\beta\delta $.
\begin{definition}
	(Homomorphism) For alphabets $ \Sigma, \Pi $, a \textit{homomorphism} $ h: \Sigma^* \to \Pi^* $ satisfies $ h(\alpha\beta) = h(\alpha)h(\beta) $ for all $ \alpha,\beta\in \Sigma^* $.
\end{definition}
\begin{remark}
  This implies that $ h(\varepsilon) = \varepsilon $. 
\end{remark}
\begin{remark}
  Alternatively we can just consider a function $ f : \Sigma \to \Pi^*$ which defines a unique homomorphism extending $ f $ by concatenation.
\end{remark}
\begin{definition}
	(Formal Langauge) A \textit{formal language} $ L $ over $ \Sigma $ is \textit{any} subset of $ \Sigma^* $.
\end{definition}
\section{Regular Languages and Finite-State Automata}
\subsection{Deterministic Finite-State Automata}
\begin{remark}
	Suppose we want to describe a language $ L $ over $ \Sigma $ using a finite amount of information. We could extend $ \Sigma $ with other symbols to get an alphabet $ \Pi \supset \Sigma $. and assign a word $ w\in\Pi^* $ to $ L $.
\end{remark}
We can do this for countably many $ L $. But we want $ w $ to \textit{describe} $ L $. We'll now see a countable class of languages where we can do this symbolically. We first give a computational method to determine these languages.
\begin{definition}
	(Deterministic Finite-State Automata) A \textit{Deterministic Finite-State Automata} (DFA) is a structure
	\[
	  D = (Q,\Sigma, \delta, q_0, F)
	\]
	where,
	\begin{enumerate}
		\item $ Q $ is a finite non-empty set of states;
		\item $ \Sigma $ is an alphabet;
		\item $ \delta $ is a transition function from $ Q\times\Sigma \to Q $;
		\item Starting state $ q_0\in Q $;
		\item $ F\subseteq Q $ is a set of acceptance states.
	\end{enumerate}
\end{definition}
The idea for a DFA is that we input a word $ w\in \Sigma^* $ we call $ w = x_1\cdots x_n $. The DFA starts in state $ q_0 $ and reads the first letter of the input string $ x_1 $. The DFA then transitions according to $ \delta $ to $ q_1 = \delta(q_0,x_1)\in Q $. We then repeat this process with $ x_2 $ and so on. Once the string is exhusted the string is accepted if $ q_n\in F $ otherwise it is rejected. This defines a language.\par
We describe these with tables and diagrams.
\begin{definition}
	(Transition diagram) A \textit{transition diagram} for a DFA, $ D $, is a directed labelled graph with:
	\begin{enumerate}
		\item The vertices being the states $ Q $.
		\item For each $ (q,\sigma)\in Q\times \Sigma $, we draw a directed edge from $ q $ to $ \delta(q, \sigma) $ and label this edge with $ \sigma $. So each vertex has $ |\Sigma| $ out-edges for every vertex.
		\item Add an arrow from outside the diagram to $ q_0 $ labelled \verb&START&.
		\item Draw vertex as a circle or as a double circle if the state is an accept state.
	\end{enumerate}
\end{definition}
If we have lots of edges from one state to another vertex, we can join them in a single edge with the label subset of $ \Sigma $.

\begin{definition}
	(Transition table) For a DFA we can also show the corresponding information in a table. The columns are symbols in $ \Sigma $ and the rows are states $ q_0, q_1, \dots $ and the $ (i,j) $ entry is $ \delta(q_i,\sigma_j) $.
\end{definition}
\subsection{Regular Languages}
\begin{definition}
	(Extended transition function) The \textit{extended transition function} of a DFA, $ D $, is
	\[
	  \hat\delta: Q \times \Sigma^* \to Q
	\]
	defined inductively via
	\[
		\hat \delta(q, \varepsilon) = q, \quad \hat\delta(q, \sigma_1) = \delta(q, \sigma_1), \quad \delta(q, \sigma_1\cdots \sigma_k) = \delta(\hat\delta(q,\sigma_1\dots \sigma_{k-1}), \sigma_k)
	\]
	for all $ q\in Q $ and $ \sigma_1,\dots, \sigma_k\in \Sigma $.
\end{definition}
So for any word $ w\in \Sigma^* $, $ \hat\delta(q_0,w) $ is the end state with input $ w $.
\begin{lemma}
  For all $ 1\le \ell \le k $, we have 
  \begin{align*}
	  \hat\delta(q, \sigma_1\dots \sigma_k) = \hat\delta( \hat\delta(q,\sigma_1 \dots\sigma_\ell), \sigma_{\ell +1} \dots \sigma_k).
  \end{align*}
\end{lemma}
\pf Induct on $ k $. We take the base case where $ k = \ell $ which follows from the definition. Otherwise,
\begin{align*}
	\text{LHS} & = \delta(\hat\delta(q, \sigma_1 \dots \sigma_{k-1}), \sigma_k)\\
		   &= \delta(\hat\delta(\hat\delta(q,\sigma_1\dots \sigma_\ell), \sigma_{\ell +1}\dots \sigma_{k-1}), \sigma_k) = \text{RHS}\qed
\end{align*}
\begin{definition}
	(Regular Language) Let $ D $ be a DFA. The language $\mathcal L(D) $ of $ D $ is the set of words accepted by $ D $ namely
	\[
		\mathcal L(D) = \left\{ w \in \Sigma^* \mid \hat\delta (q_0, w) \in F\right\} \subseteq \Sigma^*
	\]
	A language $ L $ is \textit{regular} if there exists a DFA $ D $ such that $ L = \mathcal L(D) $.
\end{definition}
\subsection{Non-determinism and $ \varepsilon $-transitions}
We now examine a seemingly more general construction where we can choose which state to move to (or equivalently move to many states simultaneously).
\begin{definition}
	(Non-deterministic Finite-State Automaton) A \textit{non-deterministic finite-state automaton} (NFA) is a structure
	\[
	  N = (Q, \Sigma, \delta, q_0, F)
	\]
	just as a for a DFA, however $ \delta $ is now a map $ Q\times \Sigma \to \mathcal P(Q) $, to contain the idea of moving to many states "at once".
\end{definition}
An NFA as input $ w= \sigma_1\dots \sigma_k \in \Sigma^* $ while in state $ q_0 $, reads $ \sigma_1 $ then "moves to" all of the states in $ \delta(q_0,\sigma_1) $. Then for all states $ p\in \delta(q_0,\sigma_1) $, read $ \sigma_2 $ and move to $ \sigma(p,\sigma_2) $. Continue to end of $ w $ to get to all end states $ E_w\subseteq Q $. If $ E_w\cap F \ne \emptyset $, then $ w $ is accepted, otherwise $ w $ is rejected.
\begin{remark}
	Note that since $ \emptyset \in\mathcal P(Q) $ we can have $ \delta(p, \sigma) = \emptyset $. But what if we're currently in states $ S\subseteq Q $ and to read $ \sigma_i $ only to find that $ \bigcup_{p\in S} \delta(p,\sigma_i) = \emptyset $. Then we are in "no state" and we are stuck in NFA limbo until the end of the word, where the word is automatically rejected.
\end{remark}
Given an NFA $ N $, we can define the transition diagram and the transition table in the same way as for a DFA although the table now contains subsets of $ \Sigma $ and the diagram has multiple directed edges for a single letter in $ \Sigma $.
\begin{definition}
	(Extended transition function) For an NFA the \textit{extended transition function} $ \hat\delta $ of $ N $ is $ \hat\delta : Q \times \Sigma^* \to \mathcal P(Q) $ defined by $ \hat\delta(q,\varepsilon) = \{q\} $ for $ q\in Q $, $ \hat\delta(q,\sigma) = \delta(q, \sigma) $ for $ q\in Q $ and $ \sigma\in \Sigma $ and
	\[
		\hat\delta(q,\sigma_1\dots\sigma_k) = \bigcup_{p\in \hat\delta(q, \sigma_1\dots \sigma_{k-1})}\delta(p,\sigma_k).
	\]
	So for $ w\in \Sigma^* $ we have that $ \hat\delta(q_0,w) $ is the set of end states $ E_w $.
\end{definition}
We want to define a more general type of state machine. We want a more flexible type where we can change certain states without reading a symbol. We think about this as reading $ \ep $ between the letters.
\begin{definition}
	($ \ep $-NFA) An $ \ep $\textit{-NFA} is a structure $ E = (Q,\Sigma, \delta, q_0, F) $ defined just as for an NFA except
	\[
		\delta: Q \times (\Sigma \cup \{\ep\}) \to \mathcal P(Q)
	\]
	where $ \ep\notin \Sigma $ but is acting like a symbol.
\end{definition}
\begin{definition}
	($ \ep $-closure) For an $ \ep $-NFA, E, and a state $ q\in Q $, we define the $ \ep $\textit{-closure} $ \mathrm{ecl}(q) $ of $ q $ as the set of states reached from $ q $ by repeated $ \ep $-transitions.
\end{definition}
That is define inductively, $ S_0(q) = \{q\} $, $ S_1 = \{q\} \cup \delta(q,\ep) $ and $ S_{i+1} (q) = S_i(q) \cup \bigcup_{r\in S_i(q)} \delta(r,\ep) $. Then we can say that
\[
	\mathrm{ecl}(q) = \bigcup_{i=0}^\infty S_i(q) = S_N(q)
\]
for some $ N $ since we have a strictly increasing chain which balances at some point. We can also take the $ \ep $-closure of a subset $ S \subseteq Q $ defined as $ \mathrm{ecl}(S) $, defined as the union of the $ \ep $-closure of each element in $ S $. We have that the $ \ep $-closure moves in and out of unions. \par
\begin{definition}
	($ \ep $-closed) We say that $ R\subseteq Q $ is $ \ep $\textit{-closed} if $ \mathrm{ecl}(R) = R $.
\end{definition}
\begin{lemma}
	For any $ S\subseteq Q $ we have that $ \mathrm{ecl}(S) $ is $ \ep $-closed.
\end{lemma}
\pf Let $ C = \mathrm{ecl}(S) $. We need that $ \mathrm{ecl}(C)\subseteq C $. But $ x\in \mathrm{ecl}(C) $ means that $ x\in \mathrm{ecl}(r) $ for some $ r\in C $, so there exists a finite sequence of $ \ep $-transitions from $ r $ to $ x $. But $ r \in \mathrm{ecl}(S) $ so there exists some $ s\in S $ with a sequence of $ \ep $-transitions from $ s $ to $ r $. Putting these together we get that $ x\in S_{m+n}(s)\subseteq \mathrm{ecl}(s)\subseteq C $, where $ m $ and $ n $ are the lengths of the $ \ep $-transitions of the paths from $ r $ to $ x $ and $ r $ to $ s $ respectively.\qed 
\par
An $ \ep $-NFA has input $ w=\sigma_1\cdots \sigma_k\in \Sigma^* $. From $ q_0 $ it moves to all states in $ \mathrm{ecl}(q_0) $ and only then reads $ \sigma_1 $. For each $ x\in\mathrm{ecl}(q_0) $ it takes the set of the states $ \delta(x,\sigma_1) $ and for each $ y_x\in \delta(x,\sigma_1) $ it now goes to $ \mathrm{ecl}(y_x) $, so now in these states unioned over $ y_x $ and $ x $. Continue with $ \sigma_2 $ and repeat, until we read the last letter $ \sigma_k $ and then $ w $ is accepted if any of these end states are in $ F $, otherwise it is rejected.
\begin{definition}
	(Extended transition function) For an $ \ep $-NFA, its \textit{extended transition function} is a function
	\[
	  \hat\delta: Q\times \Sigma^* \to \mathcal P(Q)
	\]
	defined inductvely on the length of the word $ w = \sigma_1 \cdots \sigma_k $ via $ \hat \delta(q,\ep) = \mathrm{ecl}(q) $.
	\[
		\hat\delta(q,\sigma) = \mathrm{ecl}\left(\bigcup_{p\in \mathrm{ecl}(q)} \delta(p,\sigma)\right),\quad \hat\delta(q,\sigma_1\dots \sigma_k) = \mathrm{ecl}\left( \bigcup_{p\in \hat \delta(q,\sigma_1\dots \sigma_{k-1})} \delta(p,\sigma_k)\right).
	\]
\end{definition}
What is the relationship between these finite-state machines? 
\begin{enumerate}
	\item Any DFA is a NFA (by changing the transition function to map to the singleton set insteaed).
	\item Any NFA is an $ \ep $-NFA by defining $ \delta(q,\ep) = \emptyset $.
\end{enumerate}
However all of these state machines actually define the same subset of languages called \textit{regular languages}! We'll work up to a proof of that now.
\begin{definition}
	(Language of NFAs and $ \ep $-NFAs) Given an NFA or a $ \ep $-NFA, E, it's language $ \mathcal L(E) $ is given by
	\[
		\mathcal L(E) = \left\{ w\in \Sigma^* : \hat\delta(q_0,w)\cap F\neq \emptyset \right\}.
	\]
\end{definition}
\subsection{Equivalence of DFAs and ($ \ep $-)NFAs}
Any regular language is the language of an ($ \ep $-)NFA which is fairly simple to see. We want ot show the converse using the \textit{subset construction}
\par
First suppose we have an NFA, $ N $. We produce a DFA $ D $ on the same alphabet $ \Sigma $ such that $ \mathcal L (N) = \mathcal L (D) $.
\begin{definition}
	(Subset construction) For the NFA $ N = (Q,\Sigma, \delta, q_0, F) $, the \textit{subset construction} for $ N $ produces the DFA 
	\[
		D = (\mathcal P(Q), \Sigma, \Delta, \{q_0\}, \mathcal F)
	\]
	where $ \mathcal F = \{ S \subseteq Q : S \cap F \ne \emptyset \} $ and for $ S\subseteq Q, \sigma \in \Sigma $ we have that
	\[
		\Delta(S, \sigma) = \bigcup_{p\in S} \delta(p,\sigma) \in \mathcal P(Q).
	\]
\end{definition}
\begin{theorem}
	For the NFA $ N $ and DFA $ D $ obtained from the subset construction, let $ \hat \delta $ and $ \hat \Delta $ to be their respective extended transition functions. Then for all $ w\in \Sigma^* $ we have that
	\[
		\hat \Delta(\{q_0\}, w) = \hat \delta(q_0,w).
	\]
\end{theorem}
\pf Induct on word length $ k  = | w | $. The base case $ k = 0 $ is $ w = \ep $ then the LHS $ = \{q_0\} = $ RHS. Now the inductive hypothesis
\[
	\hat \Delta (\{q_0\}, \sigma_1\dots \sigma_{k-1}) = \hat\delta( q_0,\sigma_1\dots \sigma_{k-1}).
\]
Then for $ w = \sigma_1\dots \sigma_k $, we have that
\begin{align*}
	\hat \delta(q_0,w) = \bigcup_{p\in \hat\delta(q_0,\sigma_1\dots \sigma_{k-1})} \delta(p,\sigma_k) &= \Delta(\hat\delta(q_0,\sigma\dots \sigma_{k-1}),\sigma_k) \\
													  &= \Delta(\hat \Delta(\{q_0\}, \sigma_1\dots \sigma_{k-1}),\sigma_k) = \hat\Delta(\{q_0\}, \sigma_1\dots \sigma_k)\qed\\
\end{align*}
\begin{corollary}
  A language $ L $ is accepted by some DFA if and only if it accepted by some NFA.
\end{corollary}
\pf Forwared direction is obvious. In the converse, suppose we're given an NFA $ N $, form a DFA $ D $ from the subset construction. Then $ w $ is accepted by $ N $ if and only if $ \hat \Delta(\{q_0\}, w) \in \mathcal F $ by the previous theorem.\qed
\begin{definition}
	(Subset construciton) For the $ \ep $-NFA, $ E = (Q,\Sigma, \delta, q_0, F) $ the \textit{subset construction} with $ \ep $-transitions produces the DFA $ D(\mathcal P(Q), \Sigma, \Delta, \mathrm{ecl}(\{q_0\}),\mathcal F) $ where $ \mathcal F = \{ S\subseteq Q | S\cap F \ne \emptyset \} $ as before and for $ S\subseteq Q, \sigma \in \Sigma $ we have that
	\[
		\Delta(S,\sigma) = \mathrm{ecl}\left(\bigcup_{p\in S}\delta(p,\sigma)\right)\in \mathcal P(Q).
	\]
\end{definition}
\begin{remark}
  This can be put into words as the subsets of states reached from $ S $ by following an edge labelled $ \sigma $ then some edges labelled $ \ep $. 
\end{remark}
As before we have a theorem proving this constructions defines the same language.
\begin{theorem}
  For an $ \ep $-NFA, $ E $, and DFA, $ D $, obtained from the subset construction, we have that
  \[
	  \hat \Delta(\mathrm{ecl}(\{q_0\}), w) = \hat\delta(q_0,w).
  \]
\end{theorem}
\pf As in the previous theorem, the base case for $ w\in \ep $ gives $ \mathrm{ecl}(\{q_0\}) $. Then the inductive steps is similar to the previous proof with the $ \mathrm{ecl} $ places and handled with appropiately.\qed 
\begin{corollary}
  A language $ L $ is accepted by some $ DFA $ if and only if it is accepted by some $ NFA $ if and only if it is accepted by some $ \ep $-NFA.
\end{corollary}
\subsection{Regular Expressions}
We know now that DFAs and ($ \ep $-)NFAs defnie the same languages. We knwo want to give an algebraic or symbolic description of a regular language: that of a regular expression. Then we will show that DFAs $ \implies $ Regular Expressions $ \implies $ $ \ep $-NFAs, which will show they're all equivalent.
\begin{definition}
	(Creating new languages from old)
	\begin{enumerate}
		\item The \textit{union} of languages $ L,M $ over $ \Sigma_L, \Sigma_M $ is the language $ L \cup M $ over $ \Sigma_L \cup \Sigma_M $.
		\item The \textit{concatenation} $ LM $ of languages $ L $, $ M $ is the words $ w=uv $ where $ \{(u,v) \in L\times M\}\subseteq (\Sigma_L \cup \Sigma_M)^* $.
		\item The \textit{closure} $ L^* $ of a language $ L $ is a set of all words formed by concatenating an arbitrary number of wordes in $ L $ so
			\[
				L^* = \bigcup_{n\ge 0} L^n
			\]
			including the empty word.
	\end{enumerate}
\end{definition}
\begin{remark}
  For $ n\ge 2 $ we can have $ L\not\subseteq L^n $ if $ \ep \notin L $, but not if $ \ep \in L $ since we can use the empty word to pad the word out. However certainly $ L\subseteq L^* $.
\end{remark}
\begin{definition}
	(Regular Expression) A \textit{regular expression} $ R $ is a finite string of symbols denoting a language $ \mathcal L(R) $ over some alphabet, build up from the following inductive rules:
	\begin{enumerate}
	\item Both $ \ep $ and $ \emptyset $ are regular expressions with $ \mathcal L(\ep) = \{\ep\} $ and $ \mathcal L(\emptyset) = \emptyset $.
	\item If $ a $ is a symbol then $ a $ is a regular expression with $ \mathcal L(a) = \{a\} $.
	\item If $ E $ and $ F $ are regular expressions then $ E+F $ is a regular expression denoting union so $ \mathcal L(E+F) = \mathcal L(E) \cup \mathcal L(F) $.
	\item $ EF $ is a regular expression denoting concatenation, so $ \mathcal L (EF) = \mathcal L(E)\mathcal L(F) $.
	\item $ E^* $ is a regular expression denoting the closure of $ E $ so $ \mathcal L(E^*) = \mathcal L(E)^* $.
	\item $ (E) $ is a regular expression with the same language of $ E $ and brackets are used to avoid ambiguity and enforce order of operations. 
	\end{enumerate}
\end{definition}
\begin{definition}
	(Order of operations) The order of precendence for regular expressions is brackets, then Kleene star, then concatenation, then addition.
\end{definition}
\begin{example}
	For example the regular expression $ 01^* + 1 $ defines the language accepting $ 1 $ or anything of the form $ 0(\text{any number of 1s}) $.
\end{example}
\subsubsection{Regular expressions define regular languages}
\begin{theorem}
  For each regular expression $ R $, there exists an $ \ep $-NFA, $ E $, such that $ \mathcal L(E) = \mathcal L(R) $.
\end{theorem}
\pf We build $ E $ up by induction. Our $ \ep $-NFA will be constructed such that
\begin{enumerate}
	\item There is a unique accepting state;
	\item There are no directed edges into the start state except \verb&START&;
	\item No directed edges out of the accepting state.
\end{enumerate}
We can construct $ \ep $-NFAs for the regular expressions $ \ep $, $ \emptyset $ and $ a $ just by taking two states (with the first one the starting state and the second state accepting) and joining them by $ \varepsilon $, nothing, and $ a $ respectively. \par
Now assume for a regular expression $ R $ we have an $ \ep $-NFA	 $ E $ with $ \mathcal L(E) = \mathcal L(R) $ with the construction properties as above. Now suppose we're given two regular expressions $ R $ and $ S $ each with their own $ \ep $-NFAs of the form we like. Then we can joint the $ \ep $-NFAs together by adding a new start state with $ \ep $-transitions to the two NFAs start states and then connect the NFAs end states to a new accept state also with an $ \ep $-transition. This accepts the language $ \mathcal L(R+S) $ exactly. \par
Similarly for $ RS $ we can construct a $ \ep $-NFA by doing $ R $ then $ S $ connnected by an $ \ep $-transition.
\par
Finially we can create an $ \ep $-NFA for the Kleene star, by connecting the start and end state to eachother with $ \ep $-transitions. \qed
\begin{theorem}
  For each DFA $ D $ there is a regular expression $ R $ such that $ \mathcal L(D) = \mathcal L(R) $.
\end{theorem}
\pf Given $ D = (Q,\Sigma, \delta, q_0, F) $, label the states by $ \{1, 2,\dots, n\} $ and let the start state be $ 1 $. Now given $ 1\le i,j\le n $ and $ 0\le k\le n $ we construct by induction on $ k $ a regular expression $ R_{ij}^{(k)} $ with language exactly the words $ w $ that begin at state $ i $, end at state $ j $ and are such that all the intermediate states pass through (not including the endpoints) are at most $ k $ (by label not by count). So we're thinking of $ w $ as a directed path from $ i $ to $ j $.\par
For $ k = 0 $ we have to go directly from $ i $ to $ j $. If $ i\ne j $ then set
\[
	R^{(0)}_{ij} = \begin{cases}
		a_1 + \dots + a_\ell & \ell > 0 \\
		\emptyset &\ell = 0
	\end{cases}
\]
and if $ i=j $ then let
\[
	R^{(0)}_{ii} = \begin{cases}
		a_1 + \dots + a_\ell + \ep & \ell > 0 \\
		\ep & \ell = 0
	\end{cases}
\]
where we're using $ \ep $ to denote not moving. Suppose that that $ R^{(t)}_{ij} $ is defined for all $ t<k $. Give a path $ \pi $ from $ i $ to $ j $ not passing through states $ k+1, k+2, \dots, n $. Either
\begin{enumerate}
	\item If $ \pi $ does not pass through $ k $ either, the word which would read from $ \pi $ is already in $ \mathcal L(R^{(k-1)}_{ij}) $ and vice-versa.
	\item Otherwise $ \pi $ does pass through $ k $ various times, so we split the path into $ \sigma, \pi_1,\pi_2, \dots, \pi_r, \omega $ where $ \sigma $ goes from $ i $ to $ k $, hitting $ k $ for the first time on the last step, $ \pi_\ell $ is a loop from $ k $ to $ k $ and $ \omega $ is a path from $ k $ to $ j $ which doesn't hit $ k $ (except at the start). Now the word read from $ \sigma $ is in $ \mathcal L(R_{ik}^{(k-1)}) $ and then each word $ \pi_1,\dots, \pi_r $ is in $ \mathcal L(R_{kk}^{(k-1)}) $ and $ \omega $ is in $ \mathcal L(R^{(k-1)}_{kj}) $. The the word read from $ \pi $ is in the language given by the regular expression
		\[
			R^{(k-1)}_{ik} {R^{(k-1)}_{kk}}^* R^{(k-1)}_{kj}
		\]
\end{enumerate}
This for cases (i) and (ii) we have $ \mathcal L(R^{(k-1)}_{ij} +	R^{(k-1)}_{ik}{ R^{(k-1)}_{kk}}^* R^{(k-1)}_{kj}) $. We therefore define the regular expression as $ R^{(k)}_{ij} $. Finally if $ F = \{ j_1,\dots, j_s\} $ then
\[
	\mathcal L(D) = \mathcal L(R^{(n)}_{1j_1} + \dots + R^{(n)}_{1j_s}).
\]
So we're done.\qed
\subsection{Closure Properties}
\begin{theorem}
  Let $ L,M $ be regular languages over $ \Sigma_L, \Sigma_M $. Then
  \begin{enumerate}
	  \item $ L \cup M $ is regular over $ \Sigma_L \cup \Sigma_M $;
	  \item $ LM $ is regular;
	  \item $ L^* $ is regular over $ \Sigma_L $.
  \end{enumerate}
\end{theorem}
\pf We know a language is regular if and only if it is represented by a regular expression. Let $ R_L, R_M $ be regular expressions for $ L $ and $ M $. Then we can prove as follows:
\begin{enumerate}
	\item $ R_L+R_M $ clearly works;
	\item $ R_LR_M $ clearly works;
	\item $ R_L^* $ clearly works.
\end{enumerate}
These can be checked trivally. \qed
\begin{theorem}
  Let $ L $ be a regular language over $ \Sigma $ then so is 
  \[
    \overline L = \Sigma^* \setminus L
  \]
  called the complement of $ L $.
\end{theorem}
\pf Let $ D $ be a DFA for $ L $. Then modify the DFA so every accept state is not longer an accept state and every non-accept state is an accept state, so replace $ F $ with $ Q\setminus F $. This is a DFA which clearly accepts the langauge $ \overline L $.\qed
\begin{corollary}
  If $ L,M $ are regular languages then so is $ L\cap M $.
\end{corollary}
\pf Use De Morgan's law. Since $ \overline L $ and $ \overline M $ are regular, then so is $ \overline L \cup \overline M $. Hence the complement of this is just $ L \cap M $ by De Morgan's law. \qed
\subsection{The Pumping Lemma}
How do we show a language is \textit{not} regular? We know that
\begin{enumerate}
	\item Any finite language is regular;
	\item As $ \Sigma^n $ is a finite set, if $ L $ is infinite then for all $ N $ there exists a word $ w\in L $ with $ |w| \ge N $.
\end{enumerate}
Suppose a word $ w $ is accepted by a DFA $ D $ but $ |w|\ge |Q| $. Then $ w $ defines a directed path in a transition diagram $ I'_D $ which must visit some state at least twice. So we can write $ w = xyz $ for subwords $ x,y,z $ where $ y $ gives us a loop and then go around the loop again or not at all.
\begin{theorem}
	(Pumping Lemma) If $ L $ is a regular language then there exists a $ n\ge 1 $ called the \textit{pumping number} for $ L $ such that for all words $ w\in L $ with $ |w|\ge n $, we can write $ w=xyz $ for subwords $ x,y,z $ where
	\begin{enumerate}
		\item $ y\ne \ep $;
		\item $ |xy|\le n $;
		\item $ \forall k\ge 0 $ we have the word $ xy^kz\in L $.
	\end{enumerate}
	i.e we can pump appearance of $ y $.
\end{theorem}
\pf We have that $ L=\mathcal L(D) $ for a DFA $ D = (Q,\Sigma,\delta,q_0,F) $ and set $ n=|Q| $. Now take an accepted word $ w\in L $ with $ w = \sigma_1\dots\sigma_m $ where $ m\ge n $. Let $ p_i = \hat\delta(q_0, \sigma_1\dots \sigma_i) $: the state we're in after reading the first $ i $ letters of $ w $ with $ 0\le i \le m $. By the pigeonhole principle, we must have $ 0\le \ell < r\le n $ with $ p_\ell = p_r $. So we can write \[ 
	w = \underbrace{\sigma_1\dots \sigma_\ell}_{x}\underbrace{\sigma_{\ell+1}\dots \sigma_r}_y\underbrace{\sigma_{r+1}\dots \sigma_m}_z = xyz
\]
thus we have that $ |xy| =r \le n $ and since $ \ell<r $ we have that $ y\ne \ep $. So for any $ k\ge 0 $, let's input $ xy^kz $ into $ D $. If $ k=0 $ then we can see that this is accepted since we just don't "do the loop" which $ y $ does. Similarly for $ k>1 $, the route corresponds to just doing the $ y $ loop $ k $ times which is obviously still accepted.\qed
\begin{example}
	If $ L $ is regular and infinite then the pumping lemma applies to some word $ w\in L $ so $ \{|w|: w\in L\} $ contains an arithmetic progression. Hence the language $ \{a^n: n \text{ is prime}\} $ is not regular.
\end{example}
\begin{example}
	The language $ L = \{0^i1^i: i\ge 0\} $ is \text{not} regular. Suppose that $ L $ is regular. Take $ w = 0^n1^n\in L $ for the $ n $ in the pumping lemma and write $ w=xyz $ so $ xy=0^m $ and $ y^\ell $ for some $ 0<\ell \le m\le n $. Now we can pump down, so $ xz=0^{n-\ell}1^n $ which clearly isn't in the language, but the pumping lemma says it is in the language. Alternatively we can pump up, so $ xy^2z =  0^{n+\ell}1^n $ which also clearly isn't in the language, but again the pumping lemma say it's in the language so $ L $ isn't regular.\par
	This exact argument words for $ L' = \{w: w \text{ contains same number of } 0\text{s and } 1\text{s} \}$.
\end{example}
\subsection{Equivalence relations and minimisation of DFAs}
Given a DFA $ D $ over an alphabet $ \Sigma $, we want a DFA $ D' $ over $ \Sigma $ with $ \mathcal L(D') = \mathcal (D) $ but with the smallest possible number of states. DFAs from the subset construction often have an excess of states. 
\begin{definition}
	(Accessible state) An \textit{accessible state} $ q\in Q $ of a DFA is one where $ \exists w\in \Sigma^* $ with $ \hat \delta_D(q_0,w) = q $. Otherwise we say that the state is \textit{inaccessible}.
\end{definition}
From $ D $ we can form a DFA $ A $ with no inaccessible states and $ \mathcal L(A) = \mathcal L(D) $ if $ n = |Q_D| $ then form $ S_i\subseteq Q_D $ via 
\[
	S_0 = \{q_0\},\quad S_i = \bigcup_{q\in S_i} \left(\bigcup_{\sigma\in \Sigma} \delta_D(q,\sigma)\right)
\]
so $ S_i $ is states reachable from $ q_0 $ after $ i $ steps. But any accessible state can be reached after $ n-1 $ steps (remove closed loops of repeated states) so set
\[
	S = \bigcup_{j=0}^{n-1} S_j
\]
and $ Q_A = S, F_A=F_D\cap S $ and $ \delta_A $ is a restriction of $ \delta_D $  to $ Q_A\times \sigma $ maps to $ Q_A $ by construction.
\begin{remark}
  This might not be enough. Moreover what about uniqueness?
\end{remark}
\begin{definition}
	(Equivalence) For $ A $, $ B $ both DFAs over $ \Sigma $ we say that $ A $ and $ B $ are \textit{equivalent}, written $ A\equiv B $ if (up to relabling of non-start states) they are the same DFA.
\end{definition}
\begin{remark}
  Obviously this means they define the same language too.
\end{remark}
Now for a single DFA, we say group together equivalent states.
\begin{definition}
(Equivance of states) For a DFA, $ D $, we call two states, $ p,q\in Q $ \textit{equivalent} or \textit{indistinguishable} if $\forall w \in \Sigma^* $ we have that
\[
  \hat \delta(p,w)\in F \iff \hat\delta(q,w)\in F
\]
written as $ p\sim q $, or otherwise $ p,q $ are \textit{distinguished} by $ w $.
\end{definition}
\begin{proposition}
  For a DFA, $ D $ and $ p,q\in Q $ we have that
  \begin{enumerate}
	  \item $ \sim $ is an equivalence relation on $ Q $;
	  \item If $ p\sim q $ then $ \forall\sigma \in \Sigma $ we have that $ \delta(p,\sigma) \sim\delta(q,\sigma) $ and $ \hat\delta(p,w)\sim\hat\delta(q,w) $ for all words $ w\in \Sigma^* $;
	  \item If $ p\sim q $ then $ p $ is accepting if and only if $ q $ is.
  \end{enumerate}
\end{proposition}
\pf\par
\begin{enumerate}
	\item Trivally symmetric and reflexive. For transitivity suppose that $ p\sim q $ and $ q\sim r $ and take a word $ w\in \Sigma^* $. Then both $ \hat\delta(p,w) $, $ \hat\delta(q,w) $ are either both accepting or both not accepting as are both of $ \hat \delta(q,w) $ and $ \hat\delta(r,w) $ so true for $ \hat\delta(p,w) $ and $ \hat\delta(r,w) $.
	\item Suppose we have $ \sigma \in \Sigma $ with $ \delta(p,\sigma)\not\sim \delta(q,\sigma) $ so \textit{wlog} we have $ w\in \Sigma^* $ with $ \hat\delta(\delta(p,\sigma),w) $ accepted but $ \hat\delta(\delta(q,\sigma),w) $ is not. Then $ p,q $ are distinguished by $ \sigma w $ which is a contradiction since $ p\sim q $ now induction on $ |w| $.
	\item Just put $ w=\ep $ in the definition.\qed
\end{enumerate}
We now reduce the number of states in a DFA by identifying as follows.
\begin{definition}
	(Minimal DFA) Given a DFA, $ D = (Q,\Sigma,\delta, q_0,F) $ we define the \textit{minimal DFA}, $ D/\sim $ for $ D $ as $ (Q/\sim, \Sigma, \delta, [q_0]_\sim, F') $ where
	\[
		\delta'([q],\sigma) = [\delta(q,\sigma)]
	\]
	which is well-defined by (ii) from the previous proposition and the accept states
	\[
		F' = \{[p] \mid p \in F\}
	\]
	which works by (iii) from the proposition.
\end{definition}
\begin{lemma}
  For a DFA $ D $ and $ D/\sim $ as above, take $ q\in Q $ and $ w\in \Sigma^* $. Then
  \[
	  \hat\delta'([q],w) = [\hat\delta(q,w)].
  \]
\end{lemma}
\pf Proceed by induction on $ |w| $. b
\begin{align*}
	\hat\delta'([q],\sigma_1\dots\sigma_k) &=\delta'(\hat\delta'([q],\sigma_1\dots\sigma_{k-1}), \sigma_k) \\
					       &=\delta'([\hat\delta(q,\sigma_1\dots \sigma_{k-1})],\sigma_k)=[\delta(\hat\delta(q,\sigma_1\dots\sigma_{k-1}),\sigma_k] = [\hat\delta(q,w)]\qed
\end{align*}
\begin{theorem}
  For a DFA $ D $ and its minimal DFA $ D/\sim $, we have that \[
    \mathcal L(D/\sim) = \mathcal L(D).
  \]
\end{theorem}
\pf For all $ w\in \Sigma^* $ we have that 
\begin{align*}
	w\in \mathcal L(D/\sim) &\iff \hat\delta'([q_0],w) \in F'\\
				&\iff [\hat\delta(q_0,w)] \in F' \\
				&\iff \hat\delta(q_0,w) \in F \\
				&\iff w\in \mathcal L(D).\qed
\end{align*}
\begin{lemma}
  For a DFA $ D $ and its minimal DFA $ D/\sim $, no two distinct states of $ D/\sim $ are equivalent.
\end{lemma}
\pf Suppose for two states $ p,q $ of $ D $ we have $ [p]\sim [q] $ in $ D/\sim $. Then for all words $ w $, we have $ \hat\delta'([p],q), \hat\delta'([q],w) $ are both accepting or non-accepting in $ D/\sim $. So by previous lemma $ [\hat \delta(p,q)] $, $ [\hat\delta(q,w)] $ are both accepting or non-accepting, so $ \hat\delta(p,q), \hat\delta(q,w) $ are both accepting or non-accepting in $ D $. Thus $ p\sim q $ in $ D $ and so $ [p]=[q] $ in $ D/\sim $.\qed
\begin{corollary}
  If $ D $ is a DFA then $ (D/\sim)/\sim\ \equiv D/\sim $ as DFAs.
\end{corollary}
\pf On forming $ D/\sim $, no two distinct states of $ D/\sim $ are equivalent so for each state $ s\in D/\sim $ on forming $(D/\sim)/\sim $ we see that these states are $ [s] = \{s\} $ so on relabelling back to $ s $ we find that we have some accept states and transition function for $ D/\sim $ and $ (D/\sim)/\sim $.\qed
\begin{example}
	Consider the "cat" DFA from before. This has no inaccessible states and no equivalent states. Now add a "phantom" state $ q_4 $ to $ C $ with no arrows to or from the other states and which is not an accept state then in this new DFA $ D $ we have that $ q_4 $ is inaccessible. Moreover $ \hat\delta(q_4,w)\notin F= \{q_3\} $ so $ q_4 $. Thus all states of $ D $ are inequivalent. So $ D/\sim $ is $ D $ but clearly not the smallest, since we can get rid of $ q_4 $. 
\end{example}
Here we will first remove inaccessible states, and then minimise with the equivalence.
\begin{theorem}
  If the DFA $ D $ has no inaccessible states and $ A $ is a DFA with $ \mathcal L(D) = \mathcal L(A) $ then
  \begin{enumerate}
	  \item $ A $ has at least as many states as $ D/\sim $;
	  \item If $ A $ and $ D/\sim $ have the same number of states then $ A\equiv D/\sim $.
  \end{enumerate}
\end{theorem}
\pf Take a DFA $ B $ with the fewest states, subject to $ \mathcal L(B) = \mathcal L(D) $. Then $ B $ as no inaccessible states and $ D/\sim $ also has none since for any state $ [q] $ we have some $ w $ where $ \hat\delta(q_0,w)=q $ since it holds for $ D $, then $ \hat\delta'([q_0],w) = [q] $. So we have that $ D/\sim\  = (Q/\sim, \Sigma,\delta', [q_0], F') $ and $ B=(S,\Sigma, d,S_0,G) $. \par
We form the "disjoint union DFA" $ U $ of $ D/\sim $ and $ B $ where
\[
	U = (Q/\sim\sqcup S, \Sigma, \Delta, [q_0], G\sqcup F')
\]
where
\[
  \Delta:(Q/\sim \sqcup S)\times \Sigma \to Q/\sim \sqcup S
\]
is defined via $ \delta' $ and $ d $. As $ \mathcal L(D/\sim) = \mathcal L(D) = \mathcal L(B) $, this means that states $ [q_0] $ and $ s_0 $ of $ U $ are equivalent because reading a word in $ U $ start at $ [q_0] $ is the same as it is in $ D/\sim $ and also in $ B $ starting at $ s_0 $. Then for any state $ p\in Q/\sim $, we have a state $ s\in S $ with $ p\sim s $ in $ U $ because we have some $ w\in \Sigma^* $ with $ \hat\delta([q_0],w) = p $ as it's accessible in $ D/\sim $ but $ [q_0]\sim s_0 $ in $ U $ which implies that $ p = \hat \Delta([q_0],w) \sim \tilde \Delta(s_0,w) $ in $ U=\hat d(s_0,w)=s\in S $. Now consider in $ U $. The equivalence classes of the states if $ |D/\sim |>|B| $ then have two states $ p_1,p_2 $ of $ D/\sim $ which are equivalent to a single state of $ B $ so they are equivalent to eachother in $ U $, i.e we have that
\[
  \hat \Delta(p_1,w)\in G\sqcup F' \iff \hat\Delta(p_2,w)\in G\sqcup F'.
\]
But by disjointness of the DFAs in $ U $, we have that $ \hat\Delta(p_2,w)\in G\sqcup F'\iff \hat\delta'(p_2,w)\in F' $ so $ p_1,p_2 $ are equivalent states in $ D/\sim $ which is a contradiction. So now we have $ |D/\sim|=|B| $ by minimality and each state of $ D/\sim $ is equivalent in $ U $ to a state in $ B $ and visa-versa. As pairs of equivalent statesin $ U $ ar preserved under $ \Delta $ and all states are accessible, the DFAs $ D/\sim $ and $ B $ are "matched" exactly under the equivalence, i.e $ [q_0]\sim s_0 $ in $ U $ so $ \delta'([q_0],\sigma) = \Delta([q_0], \sigma) \sim\Delta(s_0,\sigma)=d(s_0,\sigma) $ so since $ \sigma $ is arbitrary, they are the same DFA upto a relabling of states.\qed
\par
We'll now describe an algorithm to minimise a DFA. First remove inaccessible states. To find equivalent states, use \textit{Hopcroft's table filling algorithm}. Having the table $ Q\times Q $ with the $ (p,q) $th entry having a $ \times $ in it to show that $ p\not\sim q $. Only fill in the lower left part (since we have symmetry) and start with all blanks.
\par
For the base step put a $ \times $ whenver we have $ (p,q) $ with one in $ F $ and one not (since $ \ep $ distinguishes). For the inductive step take any pair $ (p,q) $ which is empty. If we have $ \sigma\in \Sigma $ with $ \delta(p,\sigma) = r $, $ \delta(q,\sigma) = s $ and $ (r,s) $ marked $ \times $. Then $ p,q $ are distinguished by $ \sigma $ concatenated with whatever distinguishses $ (r,s) $. Halt when this fails for all unmarked pairs and symbols. The algorithm works of we have a pair $ (p,q) $ unmarked by $ p\not\sim q $, take one with the shortest distinguishing word $ w= \sigma_1\dots\sigma_n $ where we say $ \hat\delta(p,w)\in F, \hat\delta(q,w)\notin F $. Now $ w\ne \ep $ (by the base step), so $ r=\delta(p,\sigma_1), s=\delta(q,\sigma_1) $ distinguished by $ \sigma_2\dots \sigma_n $, shorter word so $ (r,s) $ marked with a $ \times $ so $ (p,q) $ would be marked in the inductive step. Finally, given $ 2 $ DFAs $ D_1,D_2 $ over $ \Sigma $ we can test whether or not they have the same language. First we do the first two steps above to get $ E_1,E_2 $ then $ E_1/\sim, E_2/\sim $ which the same languages as $ D_1 $ and $ D_2 $ respectively. So $ \mathcal L(E_1/\sim) = \mathcal L(E_2/\sim) $ i.e the states are a relabling of eachother iff $ \mathcal L(D_1) = \mathcal L(D_2) $. 
\section{Context Free Grammars}
\subsection{Grammers and Languages}
\begin{definition}
	(Production rule) Given an alphabet $ \Omega $, a \textit{production} or \textit{rewrite} rule is an element $ (\alpha,\beta)\in \Omega^+\times \Omega^* $, written $ \alpha \longrightarrow \beta $, read as "replace $ \alpha $ with $ \beta $". Think of this as if $ w\in \Omega^* $ contains $ \alpha $ as a substring, so $ w = \gamma \alpha\delta $ then we can \textit{choose} to replace $ \alpha $ with $ \beta $, changing $ w $ to $ \gamma\beta\delta $.
\end{definition}
\begin{remark}
	We have that $ \alpha\ne\ep $ but we can have $ \alpha\longrightarrow \ep $ called an $ \ep $-production.
\end{remark}
\begin{definition}
	(Rewrite system) A \textit{rewrite system} $ (\Omega,P) $ is an alphabet $ \Omega $ and a finite set $ P $ of rewrite rules.
\end{definition}
\begin{definition}
	(Grammar) A (variable based) \textit{grammar} $ G = (N,\Sigma, P,S) $ is a rewrite system where $ \Omega = \Sigma \sqcup N $ is divided into $ \Sigma $ the \textit{terminal} letters and $ N $ the \textit{non-terminal} letters. Any production rule $ \alpha\to\beta\in P $ has $ \alpha\in N^+ $ (can't replace terminals). $ S\in N $ is the start symbol.
\end{definition}
\begin{remark}
  We usually use capitals for non-terminals and lower case for terminals. We use lower case greek for words over capital greek alphabets.
\end{remark}
We need to use the grammar to construct a language. For a grammar $ G = (N,\Sigma, P,S) $ and $ \alpha,\beta \in \Omega^* $ we say that 
\begin{enumerate}
	\item $ \beta $ is derivable from $ \alpha $ in one step if $ \alpha = \alpha_1\gamma\alpha_2 $, $ \beta = \alpha_1\delta\alpha_2 $ and there exists a production rule $ (\gamma,\delta) \in P $, written as
		\[
			\alpha \arr{G}{1} \beta.
		\]
	\item A \textit{derivation} of length $ n $ of $ \beta $ from $ \alpha $ written as $ \alpha\arr{G}{n}\beta $ which means that we have a $ \gamma $ with $ a\arr{G}{n-1}\gamma, \gamma\arr{G}{1} \beta $ and $ \alpha\arr G0 \alpha $.
	\item $ \alpha \arr G* \beta $ means that there exists an $ n $ with $ \alpha \arr Gn \beta $ and we say that $ \beta $ is \textit{derivable} from $ \alpha $.
	\item A word $ w\in \Omega^* $ is derivable from the start symbol $ S $ is in \textit{sequential form}.
	\item A word in sequential form is a \textit{sentence} if $ w\in \Sigma^* $ (only contains terminals).
\end{enumerate}
 \begin{definition}
	 The language $ \mathcal L(G) $ generated by a grammar G is
	 \[
		 \mathcal L(G) = \left\{w\in \Sigma^* \mid S\arr G* w\right\}
	 \]
 \end{definition} 
\begin{remark}
  In general a word $ w\in \mathcal L(G) $ could have many derivations from $ S $ of same or even different lengths. We can also have grammars which are distinct but define the same language (i.e different production rules).
\end{remark}
\subsubsection{The Chomsky hierarchy}
Suppose that $ \Omega = N\sqcup \Sigma $.
\begin{definition}
	(Chomsky Hierarchy) A production rule $ (\alpha,\beta)\in \Omega^+ \times \Omega^* $ is of
	\begin{enumerate}
		\item Type 0: If $ \alpha \in N^+ $;
		\item Type 1: Noncontracting if $ \alpha\in N^+ $ and if $ |\alpha|\le |\beta| $;
		\item Type 2: If $ \alpha = A\in N $ and $ |\beta| \ge 1 $;
		\item Type 3: If $ \alpha = A\in N $ and $ \beta = a $ or $ \beta = aB $ for $ B\in N $, $ a\in \Sigma $.
	\end{enumerate}
	We say that a grammar is of type $ n $ if all of its production rules are of type $ n $. \par
	A language $ L $ is of type $ n $ if there exists a grammar $ G $ of type $ n $ with $ L = \mathcal L(G) $.
\end{definition}
For a type 1 grammer suppose that
\[
  \beta_0 = S \arr G1 \beta_1 \arr G1\cdots \arr G1 \beta_m = w
\]
of minimum length $ m $. But $ |\beta_i|\le |\beta_{i+1}| $ and if $ \ell = |\beta_i| = \cdots = |\beta_{i+j}| $ for $ j \ge |\Omega|^l $ then $ 2 $ of these $ \beta $s are the same, so $ m $ is not minimum. Then
\[
	m\le \sum_{i=1}^{|w|} |\Omega|^\ell = M
\]
so we can tell whether or not $ w\in L=\mathcal L(G) $ systematically by creating all derivations of length $ \le M $ from $ S $ if we see $ w $ then \textit{yes} if not then \textit{no}.
\begin{remark}
  If $ \ep \in \mathcal L(G) $ then we need some $ \ep $-production $ \alpha\to\ep $ so languages of type $ 1 $ (and $ 2 $ and $ 3 $) do not contain $ \ep $.
\end{remark}
\subsection{Context free langauges}
\begin{definition}
	(Context free grammar) A \textit{context free grammar} (CFG) $ G = (N,\Sigma,P,S) $ is where every production rule is of the form $ (A,\alpha)\in N\times \Omega^* $ (so type 2 but maybe also with $ \ep $-productions).
\end{definition}
\begin{definition}
	(Context free language) A language $ L $ is \textit{context free} (CFL) if $ L = \mathcal L(G) $ for some CFG $ G $.
\end{definition}
\begin{remark}
  We write $ A\to a\mid b $ to mean the two production rules $ A\to a , A\to b $.
\end{remark}
\begin{example}
  \par
  \begin{enumerate}
	  \item A type $ 2 $ language is a context free language;
	  \item Suppose that $ N = \{S\} $, $ \Sigma = \{a,b\} $ with production rules $ S\longrightarrow aSb \mid \ep $. This generates the language $ \{a^nb^n: n\ge 0\} $ which we know is not regular by the pumping lemma.
  \item Regular languages are context free languages (Example Sheet 2, Questions 8 and 9).
  \item If we have the production rules $ S\longrightarrow SaSbS \mid SbSaS \mid \ep $ which gives us a language which has the same number of $ a $'s and $ b $'s.
  \item The langauge $ X $ which is the set of words which are palindromes is a context free language since (if we just consider $ \Sigma = \{a,b\} $) it is generated by the rules $ S\to aSa\mid bSb\mid a\mid b\mid \ep $.
  \end{enumerate}
\end{example}
